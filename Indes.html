// Fonction de génération vidéo avec réessais automatiques async function generateVideoWithRetry(prompt, maxRetries = 5) { for (let attempt = 1; attempt <= maxRetries; attempt++) { try { // Ton appel API actuel ici const response = await fetch('TON_ENDPOINT_VIDEO', { method: 'POST', headers: { 'Authorization': 'Bearer ' + taCleApi, 'Content-Type': 'application/json' }, body: JSON.stringify({ model: 'agnes-video-v2.0', prompt: prompt, // ... autres paramètres }) }); // Si erreur 500, 502, 503, 504 (serveur surchargé) if ([500, 502, 503, 504].includes(response.status)) { throw new Error(`Serveur surchargé (${response.status})`); } if (!response.ok) { const errorData = await response.json(); throw new Error(errorData.error?.message || `HTTP ${response.status}`); } return await response.json(); // Succès } catch (error) { console.warn(`Tentative ${attempt}/${maxRetries} échouée :`, error.message); if (attempt === maxRetries) { throw new Error(`Échec définitif après ${maxRetries} tentatives : ${error.message}`); } // Attente progressive (2s, 4s, 8s, 16s...) const waitTime = Math.pow(2, attempt) * 1000; console.log(`Nouvelle tentative dans ${waitTime / 1000} secondes...`); await new Promise(r => setTimeout(r, waitTime)); } } }
